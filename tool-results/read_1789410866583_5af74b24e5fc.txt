    49→router.post("auth/verify-otp", async (ctx) => {
    50→  const body = await ctx.json<{ phone: string; code: string; deviceName?: string; platform?: string }>();
    51→  const parsed = z
    52→    .object({
    53→      phone: z.string().min(6).max(20),
    54→      code: z.string().min(4).max(8),
    55→      deviceName: z.string().max(60).optional(),
    56→      platform: z.string().max(20).optional(),
    57→    })
    58→    .parse(body);
    59→  const result = await auth.verifyOtp(parsed.phone, parsed.code, parsed, ctx.ip);
    60→
    61→  const res = ok(result);
    62→  if (result.status === "ok" && result.accessToken && result.refreshToken) {
    63→    setAuthCookie(res, "sada_session", result.accessToken, 60 * 30); // 30 min sliding; client refreshes
    64→    setAuthCookie(res, "sada_refresh", result.refreshToken, 30 * 86400);
    65→  }
    66→  return res;
    67→}, { auth: false });
    68→

   132→router.post("auth/refresh", async (ctx) => {
   133→  const body = await ctx.json<{ refreshToken?: string }>().catch(() => ({ refreshToken: undefined }));
   134→  const token = body.refreshToken || ctx.cookies["sada_refresh"];
   135→  if (!token) throw ApiError.unauthorized("No refresh token");
   136→  const result = await auth.refreshSession(token, ctx.ip);
   137→  const res = ok(result);
   138→  setAuthCookie(res, "sada_session", result.accessToken, 60 * 30);
   139→  setAuthCookie(res, "sada_refresh", result.refreshToken, 30 * 86400);
   140→  return res;
   141→}, { auth: false });
